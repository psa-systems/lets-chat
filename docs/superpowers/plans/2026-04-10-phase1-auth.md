# Phase 1: Authentication Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add username/password authentication with argon2 hashing, session management via HTTP cookies, login/register pages, and route protection so only authenticated users can access the chat.

**Architecture:** Auth data lives in a separate `auth.db` SQLite database following the existing pool pattern in `db/mod.rs`. Server functions extract session cookies from the Axum request context. An `AuthLayout` component gates all chat routes, redirecting unauthenticated users to `/login`. The first registered user is auto-promoted to admin.

**Tech Stack:** Dioxus 0.7.3 fullstack, SQLite via sqlx, argon2 for password hashing, rand for token generation, axum-extra for cookie handling.

```sql
CREATE TABLE IF NOT EXISTS users (
    id            TEXT PRIMARY KEY,
    username      TEXT NOT NULL UNIQUE COLLATE NOCASE,
    display_name  TEXT,
    password_hash TEXT NOT NULL,
    role          TEXT NOT NULL DEFAULT 'user',
    is_banned     INTEGER NOT NULL DEFAULT 0,
    ban_reason    TEXT,
    banned_until  TEXT,
    is_muted      INTEGER NOT NULL DEFAULT 0,
    muted_until   TEXT,
    mute_reason   TEXT,
    created_at    TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at    TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS sessions (
    id          TEXT PRIMARY KEY,
    user_id     TEXT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    created_at  TEXT NOT NULL DEFAULT (datetime('now')),
    expires_at  TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_sessions_user ON sessions(user_id);
CREATE INDEX IF NOT EXISTS idx_sessions_expires ON sessions(expires_at);

CREATE TABLE IF NOT EXISTS invite_codes (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    code        TEXT NOT NULL UNIQUE,
    created_by  TEXT NOT NULL REFERENCES users(id),
    used_by     TEXT REFERENCES users(id),
    used_at     TEXT,
    expires_at  TEXT,
    created_at  TEXT NOT NULL DEFAULT (datetime('now'))
);
```

Note: `COLLATE NOCASE` on username makes lookups case-insensitive. `is_banned` and `is_muted` use INTEGER (SQLite's boolean).

Create `src/server_fns/auth.rs`:

```rust
use dioxus::prelude::*;

use crate::models::User;

#[server]
pub async fn register(username: String, password: String) -> Result<User, ServerFnError> {
    use argon2::{password_hash::SaltString, Argon2, PasswordHasher};
    use axum_extra::extract::cookie::{Cookie, SameSite};
    use rand::rngs::OsRng;

    let username = username.trim().to_string();
    if username.is_empty() {
        return Err(ServerFnError::new("Username cannot be empty"));
    }
    if username.len() < 3 {
        return Err(ServerFnError::new("Username must be at least 3 characters"));
    }
    if password.len() < 8 {
        return Err(ServerFnError::new("Password must be at least 8 characters"));
    }

    let pool = crate::db::get_auth_pool().await;

    // Check if username is taken
    if crate::db::auth::find_user_by_username(pool, &username)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?
        .is_some()
    {
        return Err(ServerFnError::new("Username is already taken"));
    }

    // Hash password
    let salt = SaltString::generate(&mut OsRng);
    let argon2 = Argon2::default();
    let password_hash = argon2
        .hash_password(password.as_bytes(), &salt)
        .map_err(|e| ServerFnError::new(format!("Failed to hash password: {}", e)))?
        .to_string();

    // Create user
    let user_id = crate::db::auth::create_user(pool, &username, &password_hash)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?;

    // Auto-promote first user to admin
    let user_count = crate::db::auth::count_users(pool)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?;
    if user_count == 1 {
        crate::db::auth::set_user_role(pool, &user_id, "admin")
            .await
            .map_err(|e| ServerFnError::new(e.to_string()))?;
    }

    // Create session
    let session_token = crate::db::auth::create_session(pool, &user_id)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?;

    // Set session cookie on response
    let cookie = Cookie::build(("session", session_token))
        .path("/")
        .http_only(true)
        .same_site(SameSite::Strict)
        .max_age(time::Duration::days(30))
        .build();

    let header_value = cookie.to_string().parse().map_err(|e: http::header::InvalidHeaderValue| {
        ServerFnError::new(e.to_string())
    })?;
    server_context().response_headers_mut().insert(http::header::SET_COOKIE, header_value);

    // Reload user to get correct role
    let user_record = crate::db::auth::find_user_by_id(pool, &user_id)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?
        .ok_or_else(|| ServerFnError::new("User not found after creation"))?;

    Ok(User {
        id: user_record.id,
        username: user_record.username,
        display_name: user_record.display_name,
        role: user_record.role,
        is_muted: user_record.is_muted,
        muted_until: user_record.muted_until,
        created_at: user_record.created_at,
    })
}

#[server]
pub async fn login(username: String, password: String) -> Result<User, ServerFnError> {
    use argon2::{Argon2, PasswordHash, PasswordVerifier};
    use axum_extra::extract::cookie::{Cookie, SameSite};

    let username = username.trim().to_string();
    if username.is_empty() {
        return Err(ServerFnError::new("Username cannot be empty"));
    }

    let pool = crate::db::get_auth_pool().await;

    let user_record = crate::db::auth::find_user_by_username(pool, &username)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?
        .ok_or_else(|| ServerFnError::new("Invalid username or password"))?;

    // Check ban status
    if user_record.is_banned {
        if let Some(ref until) = user_record.banned_until {
            // Check if suspension has expired
            let now = chrono::Utc::now().format("%Y-%m-%d %H:%M:%S").to_string();
            if until.as_str() > now.as_str() {
                return Err(ServerFnError::new(format!(
                    "Account suspended until {}: {}",
                    until,
                    user_record.ban_reason.as_deref().unwrap_or("No reason given")
                )));
            }
            // Suspension expired — clear it (will be done by moderation phase, for now just let them in)
        } else {
            return Err(ServerFnError::new(format!(
                "Account banned: {}",
                user_record.ban_reason.as_deref().unwrap_or("No reason given")
            )));
        }
    }

    // Verify password
    let parsed_hash = PasswordHash::new(&user_record.password_hash)
        .map_err(|e| ServerFnError::new(format!("Invalid stored hash: {}", e)))?;
    Argon2::default()
        .verify_password(password.as_bytes(), &parsed_hash)
        .map_err(|_| ServerFnError::new("Invalid username or password"))?;

    // Create session
    let session_token = crate::db::auth::create_session(pool, &user_record.id)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?;

    // Set session cookie
    let cookie = Cookie::build(("session", session_token))
        .path("/")
        .http_only(true)
        .same_site(SameSite::Strict)
        .max_age(time::Duration::days(30))
        .build();

    let header_value = cookie.to_string().parse().map_err(|e: http::header::InvalidHeaderValue| {
        ServerFnError::new(e.to_string())
    })?;
    server_context().response_headers_mut().insert(http::header::SET_COOKIE, header_value);

    Ok(User {
        id: user_record.id,
        username: user_record.username,
        display_name: user_record.display_name,
        role: user_record.role,
        is_muted: user_record.is_muted,
        muted_until: user_record.muted_until,
        created_at: user_record.created_at,
    })
}

#[server]
pub async fn logout() -> Result<(), ServerFnError> {
    use axum_extra::extract::cookie::{Cookie, SameSite};

    // Read session cookie from request
    let headers: http::HeaderMap = extract().await?;
    if let Some(cookie_header) = headers.get(http::header::COOKIE) {
        let cookie_str = cookie_header.to_str().unwrap_or("");
        for pair in cookie_str.split(';') {
            let pair = pair.trim();
            if let Some(value) = pair.strip_prefix("session=") {
                let pool = crate::db::get_auth_pool().await;
                let _ = crate::db::auth::delete_session(pool, value).await;
                break;
            }
        }
    }

    // Clear the cookie
    let cookie = Cookie::build(("session", ""))
        .path("/")
        .http_only(true)
        .same_site(SameSite::Strict)
        .max_age(time::Duration::seconds(0))
        .build();

    let header_value = cookie.to_string().parse().map_err(|e: http::header::InvalidHeaderValue| {
        ServerFnError::new(e.to_string())
    })?;
    server_context().response_headers_mut().insert(http::header::SET_COOKIE, header_value);

    Ok(())
}

#[server]
pub async fn get_current_user() -> Result<Option<User>, ServerFnError> {
    let headers: http::HeaderMap = extract().await?;
    let session_id = match headers.get(http::header::COOKIE) {
        Some(cookie_header) => {
            let cookie_str = cookie_header.to_str().unwrap_or("");
            cookie_str
                .split(';')
                .map(|s| s.trim())
                .find_map(|pair| pair.strip_prefix("session="))
                .map(|v| v.to_string())
        }
        None => None,
    };

    let session_id = match session_id {
        Some(id) if !id.is_empty() => id,
        _ => return Ok(None),
    };

    let pool = crate::db::get_auth_pool().await;
    let user_record = crate::db::auth::get_user_by_session(pool, &session_id)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?;

    Ok(user_record.map(|r| User {
        id: r.id,
        username: r.username,
        display_name: r.display_name,
        role: r.role,
        is_muted: r.is_muted,
        muted_until: r.muted_until,
        created_at: r.created_at,
    }))
}
```

    let chat_pool = crate::db::get_chat_pool().await;
    let auth_pool = crate::db::get_auth_pool().await;

#[server]
pub async fn send_message(room_id: i64, body: String) -> Result<i64, ServerFnError> {
    let body = body.trim();
    if body.is_empty() {
        return Err(ServerFnError::new("Message body cannot be empty"));
    }

    // Get current user from session
    let headers: http::HeaderMap = extract().await?;
    let session_id = headers
        .get(http::header::COOKIE)
        .and_then(|v| v.to_str().ok())
        .and_then(|cookie_str| {
            cookie_str
                .split(';')
                .map(|s| s.trim())
                .find_map(|pair| pair.strip_prefix("session="))
                .map(|v| v.to_string())
        })
        .ok_or_else(|| ServerFnError::new("Not authenticated"))?;

    let auth_pool = crate::db::get_auth_pool().await;
    let user = crate::db::auth::get_user_by_session(auth_pool, &session_id)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))?
        .ok_or_else(|| ServerFnError::new("Not authenticated"))?;

    // Check if muted
    if user.is_muted {
        if let Some(ref until) = user.muted_until {
            let now = chrono::Utc::now().format("%Y-%m-%d %H:%M:%S").to_string();
            if until.as_str() > now.as_str() {
                return Err(ServerFnError::new(format!("You are muted until {}", until)));
            }
        }
    }

    let chat_pool = crate::db::get_chat_pool().await;
    crate::db::chat::insert_message(chat_pool, room_id, &user.id, body)
        .await
        .map_err(|e| ServerFnError::new(e.to_string()))
}
